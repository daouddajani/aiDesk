-- aidesk Supabase schema
-- Run in the Supabase SQL editor (or via `supabase db push` as a migration).
-- Order matters: companies <-> profiles have a circular FK, so profiles is
-- created before the default_agent_id FK on companies is attached.

-- =========================================================================
-- Extensions
-- =========================================================================
create extension if not exists pgcrypto;   -- gen_random_uuid()
create extension if not exists vector;     -- pgvector, for suggested-answer embeddings
-- supabase_vault ships enabled on Supabase-hosted projects; used below to
-- store mailbox OAuth tokens encrypted at rest instead of a plain column.

-- =========================================================================
-- Enums
-- =========================================================================
create type user_role as enum ('super_admin', 'company_admin', 'company_agent');
create type ticket_status as enum ('new', 'pending', 'on_process', 'closed');
create type ticket_priority as enum ('low', 'medium', 'high', 'urgent');

-- =========================================================================
-- companies
-- =========================================================================
create table companies (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  logo_url text,
  timezone text not null default 'UTC',
  helpdesk_email text,
  default_agent_id uuid,           -- FK added after profiles exists, see below
  mailbox_secret_id uuid references vault.secrets (id),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- =========================================================================
-- profiles (1:1 with auth.users)
-- =========================================================================
create table profiles (
  id uuid primary key references auth.users (id) on delete cascade,
  company_id uuid references companies (id) on delete set null,
  role user_role not null default 'company_agent',
  full_name text,
  skills text[] not null default '{}',
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

alter table companies
  add constraint companies_default_agent_id_fkey
  foreign key (default_agent_id) references profiles (id) on delete set null;

create index profiles_company_id_idx on profiles (company_id);

-- =========================================================================
-- tickets
-- =========================================================================
create table tickets (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references companies (id) on delete cascade,
  subject text not null,
  sender_email text not null,
  sender_name text,
  description text not null,
  received_at timestamptz not null default now(),
  status ticket_status not null default 'new',
  assigned_agent_id uuid references profiles (id) on delete set null,
  priority ticket_priority,
  ai_suggested_agent_id uuid references profiles (id) on delete set null,
  solution_text text,
  graph_conversation_id text,       -- Microsoft Graph conversationId, for matching email replies
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  constraint tickets_closed_requires_solution
    check (status <> 'closed' or solution_text is not null)
);

create index tickets_company_id_status_idx on tickets (company_id, status);
create index tickets_assigned_agent_id_idx on tickets (assigned_agent_id);
create index tickets_graph_conversation_id_idx on tickets (graph_conversation_id);

-- =========================================================================
-- ticket_comments
-- =========================================================================
create table ticket_comments (
  id uuid primary key default gen_random_uuid(),
  ticket_id uuid not null references tickets (id) on delete cascade,
  author_id uuid not null references profiles (id) on delete restrict,
  body text not null,
  is_internal boolean not null default false,
  created_at timestamptz not null default now()
);

create index ticket_comments_ticket_id_idx on ticket_comments (ticket_id);

-- =========================================================================
-- attachments (belongs to exactly one of ticket or comment)
-- =========================================================================
create table attachments (
  id uuid primary key default gen_random_uuid(),
  ticket_id uuid references tickets (id) on delete cascade,
  comment_id uuid references ticket_comments (id) on delete cascade,
  storage_path text not null,
  filename text not null,
  mime_type text not null,
  size bigint not null,
  created_at timestamptz not null default now(),
  constraint attachments_exactly_one_parent check (
    (ticket_id is not null and comment_id is null) or
    (ticket_id is null and comment_id is not null)
  )
);

create index attachments_ticket_id_idx on attachments (ticket_id);
create index attachments_comment_id_idx on attachments (comment_id);

-- =========================================================================
-- ticket_embeddings (pgvector, for suggested-answer similarity search)
-- One row per closed ticket, embedding the problem + solution text.
-- 1536 dims matches OpenAI text-embedding-3-small; change if you swap models.
-- =========================================================================
create table ticket_embeddings (
  ticket_id uuid primary key references tickets (id) on delete cascade,
  company_id uuid not null references companies (id) on delete cascade,
  content text not null,
  embedding vector(1536) not null,
  created_at timestamptz not null default now()
);

create index ticket_embeddings_company_id_idx on ticket_embeddings (company_id);
create index ticket_embeddings_embedding_idx on ticket_embeddings
  using hnsw (embedding vector_cosine_ops);

-- =========================================================================
-- rate_limits (Postgres-backed sliding-window counter, no Redis)
-- App code upserts a row per (bucket_key, window_start) and checks `count`
-- before allowing the request through. bucket_key encodes endpoint + actor,
-- e.g. 'login:1.2.3.4' or 'ticket_comment:<user_id>'.
-- =========================================================================
create table rate_limits (
  bucket_key text not null,
  window_start timestamptz not null,
  count int not null default 1,
  primary key (bucket_key, window_start)
);

-- =========================================================================
-- updated_at trigger
-- =========================================================================
create or replace function set_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

create trigger companies_set_updated_at
  before update on companies
  for each row execute function set_updated_at();

create trigger profiles_set_updated_at
  before update on profiles
  for each row execute function set_updated_at();

create trigger tickets_set_updated_at
  before update on tickets
  for each row execute function set_updated_at();

-- =========================================================================
-- auth.users -> profiles bootstrap
-- inviteUserByEmail should pass { role, company_id, full_name } in the
-- invited user's metadata so this trigger can populate the profile row.
-- =========================================================================
create or replace function handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  insert into profiles (id, company_id, role, full_name)
  values (
    new.id,
    nullif(new.raw_user_meta_data ->> 'company_id', '')::uuid,
    coalesce((new.raw_user_meta_data ->> 'role')::user_role, 'company_agent'),
    new.raw_user_meta_data ->> 'full_name'
  );
  return new;
end;
$$;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function handle_new_user();

-- =========================================================================
-- RLS helper functions
-- security definer + fixed search_path so these can be called from RLS
-- policies on `profiles` itself without recursive-RLS deadlock.
-- =========================================================================
create or replace function current_user_role()
returns user_role
language sql
security definer
stable
set search_path = public
as $$
  select role from profiles where id = auth.uid();
$$;

create or replace function current_user_company_id()
returns uuid
language sql
security definer
stable
set search_path = public
as $$
  select company_id from profiles where id = auth.uid();
$$;

-- =========================================================================
-- Row Level Security
-- =========================================================================
alter table companies enable row level security;
alter table profiles enable row level security;
alter table tickets enable row level security;
alter table ticket_comments enable row level security;
alter table attachments enable row level security;
alter table ticket_embeddings enable row level security;
alter table rate_limits enable row level security;
-- rate_limits: no policies -> denied for `authenticated`/`anon`, only the
-- service_role (used by middleware/Server Actions) can read/write it.

-- --- companies ---
create policy companies_select on companies
  for select to authenticated
  using (
    current_user_role() = 'super_admin'
    or id = current_user_company_id()
  );

create policy companies_insert on companies
  for insert to authenticated
  with check (current_user_role() = 'super_admin');

create policy companies_update on companies
  for update to authenticated
  using (
    current_user_role() = 'super_admin'
    or (current_user_role() = 'company_admin' and id = current_user_company_id())
  );

-- --- profiles ---
create policy profiles_select on profiles
  for select to authenticated
  using (
    id = auth.uid()
    or current_user_role() = 'super_admin'
    or (current_user_role() in ('company_admin', 'company_agent')
        and company_id = current_user_company_id())
  );

create policy profiles_update on profiles
  for update to authenticated
  using (
    id = auth.uid()
    or current_user_role() = 'super_admin'
    or (current_user_role() = 'company_admin' and company_id = current_user_company_id())
  );

-- --- tickets ---
-- Super admin has no default access, per spec ("no ticket-content access
-- needed by default"): intentionally no super_admin branch here.
create policy tickets_select on tickets
  for select to authenticated
  using (company_id = current_user_company_id());

create policy tickets_update on tickets
  for update to authenticated
  using (company_id = current_user_company_id());

-- No insert/delete policy for `authenticated`: tickets are created by the
-- mailbox-polling job, which runs with the service_role key (bypasses RLS).

-- --- ticket_comments ---
create policy ticket_comments_select on ticket_comments
  for select to authenticated
  using (
    exists (
      select 1 from tickets t
      where t.id = ticket_comments.ticket_id
        and t.company_id = current_user_company_id()
    )
  );

create policy ticket_comments_insert on ticket_comments
  for insert to authenticated
  with check (
    author_id = auth.uid()
    and exists (
      select 1 from tickets t
      where t.id = ticket_comments.ticket_id
        and t.company_id = current_user_company_id()
    )
  );

-- --- attachments ---
create policy attachments_select on attachments
  for select to authenticated
  using (
    exists (
      select 1 from tickets t
      where t.id = attachments.ticket_id
        and t.company_id = current_user_company_id()
    )
    or exists (
      select 1 from ticket_comments c
      join tickets t on t.id = c.ticket_id
      where c.id = attachments.comment_id
        and t.company_id = current_user_company_id()
    )
  );

create policy attachments_insert on attachments
  for insert to authenticated
  with check (
    exists (
      select 1 from tickets t
      where t.id = attachments.ticket_id
        and t.company_id = current_user_company_id()
    )
    or exists (
      select 1 from ticket_comments c
      join tickets t on t.id = c.ticket_id
      where c.id = attachments.comment_id
        and t.company_id = current_user_company_id()
    )
  );

-- --- ticket_embeddings ---
-- Read-only from the client; rows are written server-side (service_role)
-- when a ticket is closed.
create policy ticket_embeddings_select on ticket_embeddings
  for select to authenticated
  using (company_id = current_user_company_id());
